! Title: YouTube No Shorts
! Description: Removes YouTube Shorts everywhere on the web - nav entries, shelves,
!              search results, and channel tabs. Covers desktop and mobile layouts.
! Usage: uBlock Origin (Orion) -> Settings -> Filter lists -> Import
!        AdGuard for iOS       -> Settings -> Filters -> User rules -> paste
! Note: YouTube renames its custom elements periodically. If Shorts reappear,
!       the element name has changed; the :has(a[href^="/shorts/"]) rules are the
!       most durable ones because they key off the URL, not the tag name.

! ---------- Left nav / sidebar (desktop) ----------
youtube.com##ytd-guide-entry-renderer:has(a[title="Shorts"])
youtube.com##ytd-mini-guide-entry-renderer:has(a[title="Shorts"])
youtube.com##ytd-guide-entry-renderer:has(a[href="/shorts"])

! ---------- Bottom pivot bar (mobile) ----------
m.youtube.com##ytm-pivot-bar-item-renderer:has(.pivot-shorts)
m.youtube.com##ytm-pivot-bar-item-renderer:has(a[href^="/shorts"])
m.youtube.com##.pivot-shorts

! ---------- Shorts shelves (home, subscriptions, search, channel) ----------
youtube.com##ytd-reel-shelf-renderer
youtube.com##ytd-rich-shelf-renderer[is-shorts]
youtube.com##ytd-rich-section-renderer:has(ytd-rich-shelf-renderer[is-shorts])
youtube.com##ytd-item-section-renderer:has(ytd-reel-shelf-renderer)
m.youtube.com##ytm-reel-shelf-renderer
m.youtube.com##ytm-rich-section-renderer:has(ytm-reel-shelf-renderer)
m.youtube.com##ytm-item-section-renderer:has(ytm-reel-shelf-renderer)

! ---------- Individual Shorts in feeds and search results ----------
youtube.com##ytd-video-renderer:has(a[href^="/shorts/"])
youtube.com##ytd-grid-video-renderer:has(a[href^="/shorts/"])
youtube.com##ytd-rich-item-renderer:has(a[href^="/shorts/"])
youtube.com##ytd-compact-video-renderer:has(a[href^="/shorts/"])
m.youtube.com##ytm-video-with-context-renderer:has(a[href^="/shorts/"])
m.youtube.com##ytm-compact-video-renderer:has(a[href^="/shorts/"])
m.youtube.com##ytm-rich-item-renderer:has(a[href^="/shorts/"])

! ---------- Channel page "Shorts" tab ----------
! This is the tab the native tweak could never hide cleanly.
youtube.com##yt-tab-shape:has-text(Shorts)
youtube.com##tp-yt-paper-tab:has-text(Shorts)
youtube.com##ytd-tab-renderer:has-text(Shorts)
m.youtube.com##.tab-title:has-text(Shorts)
m.youtube.com##ytm-tab-renderer:has-text(Shorts)
